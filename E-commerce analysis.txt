-- ================================================
-- TASK 1: DATA EXPLORATION
-- ================================================

-- Check total rows in each table
SELECT 'orders' as table_name, COUNT(*) as total_rows FROM orders
UNION ALL
SELECT 'customers', COUNT(*) FROM customers
UNION ALL
SELECT 'order_items', COUNT(*) FROM order_items
UNION ALL
SELECT 'products', COUNT(*) FROM products
UNION ALL
SELECT 'sellers', COUNT(*) FROM sellers;


-- ================================================
-- TASK 2: DATA CLEANING
-- ================================================

-- Check for missing values in orders
SELECT 
  COUNT(*) as total_rows,
  SUM(CASE WHEN order_id IS NULL THEN 1 ELSE 0 END) as missing_order_id,
  SUM(CASE WHEN customer_id IS NULL THEN 1 ELSE 0 END) as missing_customer_id,
  SUM(CASE WHEN order_status IS NULL THEN 1 ELSE 0 END) as missing_status
FROM orders;

-- Check for duplicate orders
SELECT order_id, COUNT(*) as count
FROM orders
GROUP BY order_id
HAVING COUNT(*) > 1;

-- Check order status breakdown
SELECT order_status, COUNT(*) as total
FROM orders
GROUP BY order_status
ORDER BY total DESC;


-- ================================================
-- TASK 3: SALES ANALYSIS
-- ================================================

-- Total revenue, orders and average order value
SELECT 
  ROUND(SUM(price), 2) as total_revenue,
  COUNT(DISTINCT order_id) as total_orders,
  ROUND(AVG(price), 2) as average_order_value
FROM order_items;

-- Monthly sales trend
SELECT 
  strftime('%Y-%m', o.order_purchase_t) as month,
  COUNT(DISTINCT o.order_id) as total_orders,
  ROUND(SUM(oi.price), 2) as monthly_revenue
FROM orders o
JOIN order_items oi ON o.order_id = oi.order_id
WHERE o.order_status = 'delivered'
GROUP BY month
ORDER BY month;

-- Top 10 best selling products by revenue
SELECT 
  oi.product_id,
  ROUND(SUM(oi.price), 2) as total_revenue,
  COUNT(*) as total_orders
FROM order_items oi
GROUP BY oi.product_id
ORDER BY total_revenue DESC
LIMIT 10;

-- Top 10 sellers by revenue
SELECT 
  oi.seller_id,
  ROUND(SUM(oi.price), 2) as total_revenue,
  COUNT(DISTINCT oi.order_id) as total_orders
FROM order_items oi
GROUP BY oi.seller_id
ORDER BY total_revenue DESC
LIMIT 10;

-- Order status summary with percentage
SELECT 
  order_status,
  COUNT(*) as total_orders,
  ROUND(COUNT(*) * 100.0 / (SELECT COUNT(*) FROM orders), 2) as percentage
FROM orders
GROUP BY order_status
ORDER BY total_orders DESC;


-- ================================================
-- TASK 4: CUSTOMER ANALYSIS
-- ================================================

-- Top 10 customers by spending
SELECT 
  c.customer_id,
  c.customer_state,
  ROUND(SUM(oi.price), 2) as total_spent,
  COUNT(DISTINCT o.order_id) as total_orders
FROM customers c
JOIN orders o ON c.customer_id = o.customer_id
JOIN order_items oi ON o.order_id = oi.order_id
GROUP BY c.customer_id, c.customer_state
ORDER BY total_spent DESC
LIMIT 10;

-- Revenue by customer state
SELECT 
  c.customer_state,
  COUNT(DISTINCT o.order_id) as total_orders,
  ROUND(SUM(oi.price), 2) as total_revenue
FROM customers c
JOIN orders o ON c.customer_id = o.customer_id
JOIN order_items oi ON o.order_id = oi.order_id
GROUP BY c.customer_state
ORDER BY total_revenue DESC
LIMIT 10;


-- ================================================
-- TASK 5: DELIVERY AND PERFORMANCE ANALYSIS
-- ================================================

-- Average delivery time in days
SELECT 
  ROUND(AVG(julianday(order_delivered) - 
  julianday(order_purchase_t)), 1) as avg_delivery_days
FROM orders
WHERE order_status = 'delivered'
AND order_delivered IS NOT NULL;

-- Cancellation rate
SELECT 
  ROUND(COUNT(CASE WHEN order_status = 'canceled' 
  THEN 1 END) * 100.0 / COUNT(*), 2) as cancellation_rate
FROM orders;

-- Average items per order
SELECT 
  ROUND(AVG(item_count), 2) as avg_items_per_order
FROM (
  SELECT order_id, COUNT(*) as item_count
  FROM order_items
  GROUP BY order_id
);
